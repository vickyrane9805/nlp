import nltk
from nltk.tag import RegexpTagger

nltk.download('punkt')
nltk.download('averaged_perceptron_tagger')

text = "The quick brown fox jumps over the lazy dog"

# Tokenization
tokens = nltk.word_tokenize(text)

print("Tokens:", tokens)

# Inbuilt POS Tagging
print("\n--- Inbuilt POS Tagging ---")
pos_tags = nltk.pos_tag(tokens)
for word, tag in pos_tags:
    print(f"{word} -> {tag}")

# Regex POS Tagging
print("\n--- Regex POS Tagging ---")

patterns = [
    (r'.*ing$', 'VBG'),
    (r'.*ed$', 'VBD'),
    (r'.*es$', 'VBZ'),
    (r'.*ould$', 'MD'),
    (r'.*\'s$', 'NN$'),
    (r'.*s$', 'NNS'),
    (r'^-?[0-9]+$', 'CD'),
    (r'.*', 'NN')
]

regexp_tagger = RegexpTagger(patterns)
regex_tags = regexp_tagger.tag(tokens)

for word, tag in regex_tags:
    print(f"{word} -> {tag}")
